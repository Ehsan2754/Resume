% Typeface: Inter (TeX Live package `inter`, OpenType files Inter-*.otf found by kpathsea), Regular + SemiBold only:
% \textbf in the content renders SemiBold, so inline emphasis reads as emphasis, not shouting.
% Why not Source Sans 3: it maps U+002D, U+00AD, U+2010 and U+2011 to one glyph, so xdvipdfmx writes U+2011 into the
% text layer ("real‑time"); ATS keyword matching and the banned-word gate (`results[- ]driven`) then miss.
% Inter has a separate glyph per hyphen code point, so "-" extracts as "-". Fallback chain: Inter -> Latin Modern Sans.
\newif\ifResumeInter \ResumeInterfalse
\IfFontExistsTF{Inter-Regular.otf}{\IfFontExistsTF{Inter-SemiBold.otf}{\IfFontExistsTF{Inter-Italic.otf}{%
  \IfFontExistsTF{Inter-SemiBoldItalic.otf}{\ResumeIntertrue}{}}{}}{}}{}
\ifResumeInter
  \defaultfontfeatures{Ligatures=TeX, Numbers=Lining, Scale=0.9}%
  \setmainfont{Inter-Regular.otf}[
    BoldFont       = Inter-SemiBold.otf,
    ItalicFont     = Inter-Italic.otf,
    SlantedFont    = Inter-Italic.otf,
    BoldItalicFont = Inter-SemiBoldItalic.otf,
    BoldSlantedFont= Inter-SemiBoldItalic.otf]
  \setsansfont{Inter-Regular.otf}[
    BoldFont       = Inter-SemiBold.otf,
    ItalicFont     = Inter-Italic.otf,
    SlantedFont    = Inter-Italic.otf,
    BoldItalicFont = Inter-SemiBoldItalic.otf,
    BoldSlantedFont= Inter-SemiBoldItalic.otf]
  % Display cut for the name: same family, tighter tracking at large size.
  \newfontfamily\ResumeDisplayFont{Inter-Regular.otf}[LetterSpace=-2.0]
\else
  \PackageWarning{resume}{Inter not found (TeX Live package inter); falling back to Latin Modern Sans}
  \let\ResumeDisplayFont\relax
\fi
