% commands.tex -- theme, type scale and title block. Single column, real text, no icons (ATS rules).
% Design system (one accent, near-black text, mid-grey metadata; hierarchy by size/weight/space, not ornament):
%   accent   #1F3A5F  "Ink navy"   -- section titles, rules, links. Nothing else is coloured.
%   text     #1D1D1F  near-black   -- all body text
%   meta     #6E6E73  mid-grey     -- dates/labels (hint column), headline, bullets, comments
%   rule     #C7CDD6  light slate  -- photo hairline
% Type scale (Inter, fontspec Scale=0.9, so effective = 0.9 x nominal): name 23.4 pt regular (tracking -2),
% headline 10.8 pt grey, section 10.8 pt semibold accent, subsection 9.9 pt semibold, body 9.9 pt
% (Inter's x-height ~0.55 em: reads like 10.5-11 pt Source Sans/Helvetica), hint column 9.0 pt grey. Bold = SemiBold.

% Section heading: accent title in the main column, a thin accent hairline spanning the main column under it.
% The hint column stays empty, so the left edge of every heading lines up with the text it introduces.
\makeatletter
% \cvitem takes short arguments (no \par), so the two-line heading body lives in its own long macro.
\newcommand{\ResumeSectionBody}[1]{\parbox[t]{\maincolumnwidth}{\strut\sectionstyle{#1}\par\vspace{-1.1ex}{\color{color1}\rule{\maincolumnwidth}{0.4pt}}}}
\RenewDocumentCommand{\section}{sm}{%
  \par\addvspace{1.7ex}%
  \phantomsection{}%
  \addcontentsline{toc}{section}{#2}%
  \cvitem[0ex]{}{\ResumeSectionBody{#2}}%
  \par\nobreak\addvspace{0.6ex}\@afterheading}
\RenewDocumentCommand{\subsection}{sm}{%
  \par\addvspace{0.8ex}%
  \phantomsection{}%
  \addcontentsline{toc}{subsection}{#2}%

% Entry: role semibold, organisation regular (no slant), location in grey; description at body size, not \small.
\renewcommand*{\cventry}[7][.3em]{%
  \cvitem[#1]{#2}{%
    {\bfseries#3}%
    \ifthenelse{\equal{#4}{}}{}{, #4}%
    \ifthenelse{\equal{#5}{}}{}{, {\color{color2}#5}}%
    \ifthenelse{\equal{#6}{}}{}{, {\color{color2}#6}}%
    \strut%
    \ifx&#7&%
    \else{\newline{}\begin{minipage}[t]{\linewidth}#7\end{minipage}}\fi}}
% Comment column of \cvitemwithcomment / \cvlanguage: grey, upright, small.
\renewcommand*{\cvitemwithcomment}[4][.25em]{%
  \savebox{\cvitemwithcommentbox}{{#3}}%
  \setlength{\cvitemwithcommentskilllength}{\widthof{\usebox{\cvitemwithcommentbox}}}%
  \setlength{\cvitemwithcommentcommentlength}{\maincolumnwidth-\separatorcolumnwidth-\cvitemwithcommentskilllength}%
  \cvitem[#1]{#2}{%
    \begin{minipage}[t]{\cvitemwithcommentskilllength}\usebox{\cvitemwithcommentbox}\end{minipage}%
    \hfill
    \begin{minipage}[t]{\cvitemwithcommentcommentlength}\raggedleft\small\textcolor{color2}{#4}\end{minipage}}}
\makeatother

% Title block: name, headline, optional photo. Signature unchanged: \myCVtitle{first}{last}{headline}{photo path or empty}.
% Photo (DE/EU builds only; photo.txt = none for US-HQ): modern German header convention -- small portrait (~24 x 32 mm,
% 3:4), top-right, top edge on the cap height of the name, no frame or padding; a 0.3 pt light-slate hairline on a
% 2 mm-radius rounded clip keeps a white studio background from dissolving into the page.
% Evidence: vault note "Benchmark -- Resumes That Landed" (bewerbung-express.de: modern layout ~3 x 4 cm, frameless,
% optional rounded crop; talentvp.com: header beside the name or top-right). tikz ships in pgf (already in TL_PACKAGES).
\RequirePackage{tikz}
\newlength{\ResumePhotoW}
\setlength{\ResumePhotoW}{24mm}
\newlength{\ResumePhotoH}
\newsavebox{\ResumePhotoBox}
\newcommand{\myPhoto}[1]{%
  \sbox{\ResumePhotoBox}{\includegraphics[width=\ResumePhotoW]{#1}}%
  \setlength{\ResumePhotoH}{\ht\ResumePhotoBox}%
  \begin{tikzpicture}[baseline=(current bounding box.north)]
    \begin{scope}
      \clip[rounded corners=2mm] (0,0) rectangle (\ResumePhotoW,\ResumePhotoH);
      \node[anchor=south west, inner sep=0pt, outer sep=0pt] at (0,0) {\usebox{\ResumePhotoBox}};
    \end{scope}
    \draw[ResumeRule, line width=0.3pt, rounded corners=2mm] (0,0) rectangle (\ResumePhotoW,\ResumePhotoH);
  \end{tikzpicture}}
\newcommand{\myName}[2]{{\ResumeDisplayFont\namefont\color{color0}#1~#2}}
\newcommand{\myCVtitle}[4]{%
 \edef\myPhotoPath{#4}%
 \noindent
 \ifx\myPhotoPath\empty
   \begin{minipage}[t]{\textwidth}
 \else
   \begin{minipage}[t]{\dimexpr\textwidth-\ResumePhotoW-6mm\relax}
 \fi
   \vspace{0pt}% top-align: the box's reference point is its top edge, not the first baseline
   \RaggedRight
   \myName{#1}{#2}\par\vspace{1.2ex}%
   {\titlefont\color{color2}\hyphenpenalty=10000\exhyphenpenalty=10000 #3\par}%
 \end{minipage}%
 \ifx\myPhotoPath\empty\else
   \hfill
   \begin{minipage}[t]{\ResumePhotoW}\vspace{0pt}\myPhoto{\myPhotoPath}\end{minipage}%
 \fi
 \par\vspace{2.0ex}%
}
